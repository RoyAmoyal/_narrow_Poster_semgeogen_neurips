\block{\huge\vphantom{Akg}Motivation \& Goal}{
\Large
\textbf{Cross-instance semantic-geometric correspondence} matches the same semantic and geometric \mbox{location} on \emph{different} objects in a category, e.g., the \emph{left} wing tip of two airplanes.

\vspace{0.3em}
\textbf{Problem:} supervision is \textbf{manual, sparse, and costly} to scale, and exact point-level matches
across instances are often ill-defined.

\vspace{0.3em}
\textbf{Our solution:} automatically generate {\setlength{\fboxsep}{0.12em}\colorbox{yellowpastel}{\strut\emph{approximate} semantic-geometric correspondences}}
from rotation-aligned 3D objects. This yields scalable 3D--3D, 3D--2D, and 2D--2D supervision
without manual labels. Used as pretraining data, it improves state-of-the-art models.

\block{\huge\vphantom{Akg}Method}{
\Large
\begin{itemize}[leftmargin=1.1em,itemsep=0.7em,topsep=0pt]
    \item \textbf{Semantic features $f$:} dense DINOv2 features from many rendered views, lifted to 3D.
    \item \textbf{Continuous piecewise-affine (CPA) registration:} deform the source onto the target
    by moving control-grid vertices by $u_v$ ($\mathcal{E}$: neighboring-vertex edges),
    \[
    \min_{T}\; \cL_{\mathrm{Chamfer}}\big(\hat{\cX}_S,\cX_T\big)+\lambda\,\mathcal{R}_{\mathrm{smooth}}(T),
    \qquad \hat{\cX}_S=T(\cX_S),
    \]
    \[
    \cL_{\mathrm{Chamfer}}=
    \tfrac{1}{|\hat{\cX}_S|}\!\sum_{\hat{x}\in\hat{\cX}_S}\min_{x\in\cX_T}\|\hat{x}-x\|_2^2
    +\tfrac{1}{|\cX_T|}\!\sum_{x\in\cX_T}\min_{\hat{x}\in\hat{\cX}_S}\|x-\hat{x}\|_2^2,
    \qquad
    \mathcal{R}_{\mathrm{smooth}}=\tfrac{1}{|\mathcal{E}|}\!\sum_{(a,b)\in\mathcal{E}}\|u_a-u_b\|_2^2,
    \]
    \item \textbf{Match \& prune:} nearest-neighbor matching after deformation; keep only matches with
    cosine similarity $\mathrm{sim}(f^S_i,f^T_j)>0.85$.
    \item \textbf{Project to 2D, scale combinatorially:} $N$ objects yield up to $\binom{N}{2}$ pairs, each rendered
    from many views into 3D--2D and 2D--2D matches.
\end{itemize}
}
