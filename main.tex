\block{\huge Motivation \& Problem}{
\Large
\textbf{Goal: category-level 3DGS alignment.}
Given two independently reconstructed Gaussian Splatting models, estimate a global similarity transform
\[
T=(s,R,\bft)\in\Sim(3)
\]
that aligns the source model $\G_1$ to the target model $\G_2$.

\vspace{0.25em}

\noindent
\begin{minipage}[t]{0.48\linewidth}
\textbf{Why it matters}
\begin{itemize}[nosep,leftmargin=1.05em]
    \item \mbox{Align different object instances.}
    \item \mbox{Enable synchronized rendering.}
    \item Support geometrically consistent object replacement.
\end{itemize}
\end{minipage}
\hfill
\begin{minipage}[t]{0.48\linewidth}
\textbf{Why it is hard}
\begin{itemize}[nosep,leftmargin=1.05em]
    \item \mbox{Arbitrary frames and scales.}
    \item \mbox{Shape variation across instances.}
    \item \mbox{Ambiguous or missing}\\
          \mbox{correspondences.}
\end{itemize}
\end{minipage}
}

\block{\huge Coarse: Feature-Guided Sim(3) Solver}{
\Large
\textbf{Idea:} extend ICP-like absolute orientation with semantic-geometric pruning, then solve a closed-form Sim(3) update.

\vspace{0.35em}

\textbf{1. Feature-guided candidate set}
\[
Q_i=\left\{\bfq_j\in\G_2\;\middle|\;
\|\bff_i-\bff_j\|_2\leq\tau_f\right\}.
\]

\textbf{2. Spatial nearest neighbor after pruning}
\[
\bfq_i^{(k)}=\arg\min_{\bfq\in Q_i}
\|\bfp_i^{(k)}-\bfq\|_2.
\]

\textbf{3. Closed-form similarity update}
\[
T^{(k)}=\arg\min_{T\in\Sim(3)}
\sum_i\left\|T(\bfp_i^{(k)})-\bfq_i^{(k)}\right\|_2^2.
\]

\vspace{0.3em}

\Large
The coarse stage is robust to poor initializations, including large rotations, scale gaps, and cross-instance shape variation.
}

\block{\huge Fine: From Inverse Radiance Fields to MVFC}{
\Large

\noindent\makebox[\linewidth][c]{\textbf{Classical inverse radiance-field pose estimation [2]}}\\[-1em]
\[
\min_{C\in\SE(3)}
\left\|
\Rend_{\rm rgb}(\G,C)-I_t
\right\|_2^2.
\]

\vspace{0.5em}

\noindent\makebox[\linewidth][c]{\textbf{Extension to multi-view field-to-field alignment}}\\[-1em]
\[
\min_{T\in\Sim(3)}
\sum_{k=1}^{N}
\left\|
\Rend_{\rm rgb}(T\G_1,C_k^\star)
-
\Rend_{\rm rgb}(\G_2,C_k^\star)
\right\|_2^2.
\]

\vspace{0.5em}

\noindent\makebox[\linewidth][c]{\textbf{Adaptation to cross-instance feature-field alignment}}\\[-1em]
\[
\mathcal{L}_{\rm MV\text{-}FC}
=
\sum_{k=1}^{N}
\left\|
\Rend_{\rm f}(T\G_1,C_k^\star)
-
\Rend_{\rm f}(\G_2,C_k^\star)
\right\|_2^2.
\]

\vspace{0em}

\[
T^\star =
\operatorname*{arg\,min}_{T\in\Sim(3)}
\mathcal{L}_{\rm MV\text{-}FC}.
\]

\vspace{0.6em}

\Large
The fine stage refines the coarse alignment by rendering both feature-augmented 3DGS models from multiple target views and optimizing the global $\Sim(3)$ transformation.
}
